19. MAP + FILTER + SORTED
============================================================

students = [
    {"name": "Anna", "score": 85},
    {"name": "Oleg", "score": 92},
    {"name": "Ira", "score": 78},
    {"name": "Max", "score": 90}
]

filtered = filter(
    lambda student: student["score"] > 80,
    students
)

names = map(
    lambda student: student["name"],
    filtered
)

result = sorted(names)

print(result)
# ['Anna', 'Max', 'Oleg']

22. ДЕКОРАТОРЫ
============================================================

def my_decorator(func):

    def wrapper():
        print("Начало")
        func()
        print("Конец")

    return wrapper


@my_decorator
def hello():
    print("Привет!")


hello()

@my_decorator означает примерно:
hello = my_decorator(hello)


------------------------------------------------------------
Декоратор с аргументами
------------------------------------------------------------

def debug(func):

    def wrapper(*args, **kwargs):
        print(f"Вызов: {func.__name__}")
        result = func(*args, **kwargs)
        print(f"Результат: {result}")
        return result

    return wrapper


@debug
def add(a, b):
    return a + b

print(add(3, 7))
